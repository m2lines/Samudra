   yi=np.flatnonzero((lat>=ylim[0])&(lat<=ylim[1]));xi=np.flatnonzero((lon>=xlim[0])&(lon<=xlim[1]))
   fig.text(.02,1-(120+row*rowh+20)/h,region,fontsize=10)
