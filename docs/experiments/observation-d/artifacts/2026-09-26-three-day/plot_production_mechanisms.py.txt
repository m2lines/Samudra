"""Plot fixed-model interventions; add scratch automatically when evidence arrives."""
import gzip
import json
from pathlib import Path
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import numpy as np

out = Path(__file__).resolve().parent
arms = [a for a in ['scratch', 'sequential', 'mixed'] if (out / f'production-mechanisms-{a}.json.gz').exists()]
leads = [5, 15, 30, 90, 180, 365]
styles = {
    'baseline': ('Unmodified', '#222222', '-'),
    'initial-hidden-seasonal': ('Initial hidden-state reset', '#009e73', ':'),
    'every-step-hidden-seasonal': ('Every-step hidden-state reset', '#d55e00', '--'),
    'every-step-velocity-seasonal': ('Every-step velocity-slot reset', '#0072b2', '-.'),
}
fig, axes = plt.subplots(3, len(arms), figsize=(5 * len(arms), 9), sharey='row', squeeze=False)
for col, arm in enumerate(arms):
    with gzip.open(out / f'production-mechanisms-{arm}.json.gz', 'rt') as stream:
        evidence = json.load(stream)
    records = list(evidence['results'].values())
    assert len(records) == 3
    for row, (field, label) in enumerate([('sst_rmse', 'SST RMSE (°C)'), ('adt_rmse', 'ADT RMSE (m)')]):
        ax = axes[row, col]
        for mode, (name, color, line) in styles.items():
            values = np.array([[r[mode]['leads'][str(day)][field] for day in leads] for r in records])
            assert np.isfinite(values).all()
            ax.plot(leads, values.mean(0), label=name, color=color, linestyle=line, marker='o', markersize=3)
        ax.set_xscale('log')
        ax.set_xticks(leads, [str(d) for d in leads])
        ax.set_xlabel('Forecast lead (days)')
        ax.grid(alpha=.18)
        if col == 0:
            ax.set_ylabel(label)
    ax = axes[2, col]
    for x, (mode, (_, color, _)) in enumerate(styles.items()):
        per_origin = np.array([np.mean([v['error_dex'] for k, v in r[mode]['spectra'].items() if k.startswith('annual-eke/')]) for r in records])
        assert np.isfinite(per_origin).all()
        ax.bar(x, per_origin.mean(), color=color, alpha=.65)
        ax.scatter(np.array([-.10, 0, .10]) + x, per_origin, color=color, s=16, edgecolors='white', linewidths=.4)
    ax.set_xticks(range(4), ['Unmodified', 'Initial\nhidden', 'Every step\nhidden', 'Every step\nvelocity'], fontsize=8)
    ax.grid(axis='y', alpha=.18)
    if col == 0:
        ax.set_ylabel('Annual EKE spectral error (dex)')
    axes[0, col].set_title('Samudra2 ' + arm)
handles, labels = axes[0, 0].get_legend_handles_labels()
fig.legend(handles, labels, loc='lower center', ncol=2, fontsize=9, bbox_to_anchor=(.5, .005))
fig.suptitle('Fixed-model interventions: lower point error can hide worse spectra', fontsize=12)
fig.tight_layout(rect=(0, .075, 1, .96))
fig.savefig(out / 'production-mechanisms.png', dpi=115)
plt.close(fig)
(out / 'production-mechanisms.png.license').write_text('SPDX-FileCopyrightText: 2026 Samudra Authors\nSPDX-License-Identifier: CC-BY-4.0\n')
